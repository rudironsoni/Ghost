📋 EMERGENCY PROCEDURE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SITUATION: Canary deployment is causing issues and needs immediate rollback

STEP 1 - QUICK START (interactive, with confirmation):
  $ cd /path/to/Ghost
  $ ./deploy/rollback.sh

STEP 2 - AUTOMATED (for CI/CD systems, no confirmation):
  $ ./deploy/rollback.sh --force

STEP 3 - MONITOR RESULTS:
  $ tail -f logs/rollback_*.log
  $ docker-compose -f deploy/docker-compose.canary.yml ps

STEP 4 - VERIFY STABLE IS SERVING:
  $ curl http://localhost/
  $ docker exec app-stable curl http://localhost:3000/health

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

⚡ WHAT ROLLBACK.SH DOES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Displays warning and gets confirmation (unless --force flag)
2. Backs up current nginx config to .rollback_backup/
3. Drains active connections from canary (30 second wait)
4. Stops canary container gracefully (or force kills if needed)
5. Verifies stable service is healthy (auto-restarts if needed)
6. Updates nginx to route 100% traffic to stable
7. Removes canary container and cleans up resources
8. Verifies rollback succeeded and stable is responding
9. Generates colored console output and detailed logs

Exit Codes:
  0 = Success ✓
  1 = Failed (needs investigation)
  2 = Invalid arguments

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🔍 IF ROLLBACK FAILS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CHECK LOGS:
  $ tail -f logs/rollback_*.log

CHECK SERVICES:
  $ docker-compose -f deploy/docker-compose.canary.yml ps

CHECK NGINX:
  $ docker exec nginx-canary curl -v http://localhost/

RESTORE BACKUP:
  $ docker cp .rollback_backup/nginx_backup_*.conf \
    nginx-canary:/etc/nginx/conf.d/default.conf
  $ docker exec nginx-canary nginx -s reload

CALL DEVOPS TEAM if:
  - Logs show critical failures
  - Stable service won't come up
  - Nginx won't reload properly
  - Script exits with error code 1
